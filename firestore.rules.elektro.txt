// ===== Elektro editor — blok pravidel pro Firestore =====
// Vložit DOVNITŘ stávajícího  match /databases/{database}/documents { ... }  v projektu vrana-denik,
// vedle pravidel Deníku. Nic z Deníku se nemění; users_auth už Deník používá.
//
//   Přístup má JEN uživatel, který má v users_auth/{uid} pole `elektro`:
//     ["*"]            = všechny zakázky (správce)
//     ["belcicka", …]  = jen vyjmenované zakázky (subdodavatel)
//   Role z Deníku (admin/worker/sub) sama o sobě přístup NEDÁVÁ.

function elektroSeznam() {
  return get(/databases/$(database)/documents/users_auth/$(request.auth.uid)).data.get('elektro', []);
}
function elektroSmi(jobId) {
  return request.auth != null && ('*' in elektroSeznam() || jobId in elektroSeznam());
}
match /elektro_zakazky/{jobId} {
  allow read: if elektroSmi(jobId);
  allow write: if request.auth != null && '*' in elektroSeznam();
  match /stav/{doc} {
    allow read, write: if elektroSmi(jobId);
  }
  match /historie/{doc} {
    allow read, write: if elektroSmi(jobId);
  }
}
